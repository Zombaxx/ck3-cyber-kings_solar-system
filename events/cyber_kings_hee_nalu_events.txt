# He'e Nalu (2026-11-15): the "Go Surfing" event chain. Design and branch map: _dev/HEE_NALU.md
# Started by the Surf Trip activity when its phase goes active at the chosen break
# (common/activities/activity_types/cyber_kings_hee_nalu_surf.txt, on_phase_active). Every event shows the beach scene.
#
# 0001 Dawn Patrol ─┬─ listen ──► 0002 The Ocean Speaks ─┬─ felt it ──► 0010 or 0020, attuned
#                   │                                     └─ nothing ──► 0010 or go home
#                   ├─ paddle out ──► 0010 The Lineup ──► 0030 / 0011 / 0023
#                   ├─ outer reef ──► 0020 The Outer Reef ──► 0021 The Drop / 0022 Held Down / 0011
#                   └─ watch from the sand (ends)
# 0023 A Shadow Beneath ──► 0030 or ends     0021 The Drop ──► ends or 0030
# 0030 The Perfect Wave, 0011 A Good Session, 0022 Held Down: endings
#
# GUESTS (player or AI): 0100 Paddling Out -> 0110 A Wave of My Own / 0111 The Washing Machine, or stay in the whitewater, or watch.
#                         Their options call hee_nalu_guest_progress_effect (trait chances and XP by outcome).
# FAITH LIFE (2026-11-16): 0200 The Paddle-Out (heir, from on_death), 0210 the house 'aumakua (decision),
#                         0301-0306 yearly flavour (common/on_action/cyber_kings_hee_nalu_on_actions.txt)
# MAKAHIKI (2026-11-16): 0400 The Games of Lono, 0403 Lono Departs (hee_nalu_proclaim_makahiki_decision)
# PILGRIMAGE & ADVENTURERS (2026-11-18): 0500 The Sacred Break (pilgrimage arrival), 0307 A Lord on the Beach (yearly,
#                         adventurers), 0600 / 0610 contract events (common/task_contracts/cyber_kings_hee_nalu_contracts.txt)
# Surf skill: hee_nalu_surf_skill_value (trait level, or -15 for a kook) is added to every roll for the better outcome;
#                         endings call hee_nalu_host_good_session_effect / hee_nalu_host_rough_session_effect (XP, traits, guests)
# scope:surf_companion  = the travelling companion you invited to the trip, if any (the activity's special guest)
# scope:surf_break      = the province you surf at; hee_nalu_sacred / _home / _foreign / _far / _new / _new_sacred describe it
#                         and feed hee_nalu_break_rewards_effect and hee_nalu_record_break_effect (common/scripted_effects)
# scope:hee_nalu_attuned = set when you sensed the ocean in 0002; raises the odds everywhere after

namespace = hee_nalu

# Dawn Patrol
hee_nalu.0001 = {
	type = character_event
	title = hee_nalu.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:hee_nalu_sacred }
				desc = hee_nalu.0001.desc.sacred
			}
			triggered_desc = {
				trigger = { exists = scope:hee_nalu_home }
				desc = hee_nalu.0001.desc.home
			}
			desc = hee_nalu.0001.desc.foreign
		}
		desc = hee_nalu.0001.desc
	}
	theme = faith
	override_background = { reference = bp3_coast }
	left_portrait = {
		character = root
		animation = personality_content
	}
	right_portrait = {
		character = scope:surf_companion
		trigger = { exists = scope:surf_companion }
		animation = happiness
	}

	immediate = {
		location = { save_scope_as = surf_break }
		if = {
			limit = { exists = scope:activity }
			scope:activity.special_guest:honorary_guest_traveling_companion ?= {
				if = {
					limit = {
						is_alive = yes
						location = root.location
					}
					save_scope_as = surf_companion
				}
			}
		}
		# what kind of break is this? (same tests as the activity's province_description)
		if = {
			limit = {
				faith = { any_holy_site_barony = { county = scope:surf_break.county } }
			}
			save_scope_value_as = { name = hee_nalu_sacred value = yes }
		}
		if = {
			limit = {
				scope:surf_break = {
					county_controller ?= {
						OR = {
							this = root
							target_is_liege_or_above = root
						}
					}
				}
			}
			save_scope_value_as = { name = hee_nalu_home value = yes }
		}
		else = {
			save_scope_value_as = { name = hee_nalu_foreign value = yes }
		}
		if = {
			limit = {
				exists = capital_province
				"capital_province.squared_distance(scope:surf_break)" > squared_distance_large
			}
			save_scope_value_as = { name = hee_nalu_far value = yes }
		}
		if = {
			limit = {
				NOT = {
					is_target_in_variable_list = {
						name = hee_nalu_breaks_surfed
						target = scope:surf_break
					}
				}
			}
			save_scope_value_as = { name = hee_nalu_new value = yes }
			if = {
				limit = {
					exists = scope:hee_nalu_sacred
					NOT = {
						any_in_list = {
							variable = hee_nalu_breaks_surfed
							county = scope:surf_break.county
						}
					}
				}
				save_scope_value_as = { name = hee_nalu_new_sacred value = yes }
			}
		}
	}

	option = { # Sit and listen to the ocean
		name = hee_nalu.0001.a
		custom_tooltip = hee_nalu.0001.a.tt
		stress_impact = {
			patient = minor_stress_impact_loss
			calm = minor_stress_impact_loss
			impatient = minor_stress_impact_gain
		}
		trigger_event = hee_nalu.0002
		ai_chance = {
			base = 35
			modifier = { add = 25 has_trait = patient }
			modifier = { add = 25 has_trait = calm }
			modifier = { add = -25 has_trait = impatient }
		}
	}

	option = { # Paddle straight out
		name = hee_nalu.0001.b
		custom_tooltip = hee_nalu.0001.b.tt
		trigger_event = hee_nalu.0010
		ai_chance = {
			base = 35
			modifier = { add = 25 has_trait = impatient }
		}
	}

	option = { # Charge the outer reef
		name = hee_nalu.0001.c
		custom_tooltip = hee_nalu.0001.c.tt
		stress_impact = {
			brave = minor_stress_impact_loss
			craven = medium_stress_impact_gain
		}
		trigger_event = hee_nalu.0020
		ai_chance = {
			base = 15
			modifier = { add = 35 has_trait = brave }
			modifier = { add = 15 has_trait = arrogant }
			modifier = { factor = 0 has_trait = craven }
		}
	}

	option = { # Watch from the sand
		name = hee_nalu.0001.d
		add_stress = minor_stress_loss
		stress_impact = {
			brave = minor_stress_impact_gain
			lazy = minor_stress_impact_loss
			craven = minor_stress_impact_loss
		}
		ai_chance = {
			base = 5
			modifier = { add = 40 has_trait = craven }
			modifier = { add = 20 has_trait = lazy }
		}
	}
}
