# Invitation into the European Confederation. root = invited ruler, scope:ck_eu_confed_founder = founder
ck_situations.0102 = {
	type = character_event
	title = ck_situations.0102.t
	desc = ck_situations.0102.desc
	theme = diplomacy
	left_portrait = scope:ck_eu_confed_founder
	trigger = {
		ck_eu_confederation_candidate_trigger = yes
		scope:ck_eu_confed_founder.house.house_confederation ?= { confederation_type = confederation_type:ck_european_confederation }
	}
	option = {	# join
		name = ck_situations.0102.a
		scope:ck_eu_confed_founder.house.house_confederation = {
			add_confederation_member_house = root.house
		}
		ck_fire_situation_catalyst_effect = { SITUATION = ck_european_question CATALYST = ck_catalyst_eu_confederation_joined CHARACTER = root }
		ai_chance = {
			base = 50
			opinion_modifier = { opinion_target = scope:ck_eu_confed_founder }
		}
	}
	option = {	# stay out
		name = ck_situations.0102.b
		ai_chance = { base = 50 }
	}
}

# From Confederation to Union: Integration began while Europe has no Union. root = leader of the European Confederation
ck_situations.0103 = {
	type = character_event
	title = ck_situations.0103.t
	desc = ck_situations.0103.desc
	theme = realm
	left_portrait = root
	trigger = { NOT = { exists = title:e_european_union.holder } }
	immediate = {
		house.house_confederation ?= { save_scope_as = ck_eu_confed }
	}
	option = {	# proclaim the Union
		name = ck_situations.0103.a
		custom_tooltip = ck_situations.0103.a.tt
		ck_eu_create_union_effect = yes
		save_scope_as = ck_eu_union_founder
		hidden_effect = {
			scope:ck_eu_confed ?= {
				every_confederation_member = {
					limit = {
						this != root
						is_independent_ruler = yes
					}
					trigger_event = ck_situations.0104
				}
			}
			scope:ck_eu_confed ?= { remove_confederation_member_house = root.house }	# the Union replaces the confederation
		}
		ai_chance = { base = 100 }
	}
	option = {	# not yet
		name = ck_situations.0103.b
		ai_chance = { base = 0 }
	}
}

# A confederate is asked to enter the new Union as a vassal. root = confederate, scope:ck_eu_union_founder = new Union ruler
ck_situations.0104 = {
	type = character_event
	title = ck_situations.0104.t
	desc = ck_situations.0104.desc
	theme = diplomacy
	left_portrait = scope:ck_eu_union_founder
	trigger = {
		is_independent_ruler = yes
		scope:ck_eu_union_founder = { has_title = title:e_european_union }
	}
	option = {	# enter the Union
		name = ck_situations.0104.a
		house.house_confederation ?= { remove_confederation_member_house = root.house }
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = change
			add_claim_on_loss = no
		}
		change_liege = {
			liege = scope:ck_eu_union_founder
			change = scope:change
		}
		resolve_title_and_vassal_change = scope:change
		ai_chance = {
			base = 50
			opinion_modifier = { opinion_target = scope:ck_eu_union_founder }
		}
	}
	option = {	# keep our sovereignty and leave the confederation
		name = ck_situations.0104.b
		house.house_confederation ?= {
			remove_confederation_member_house = root.house
		}
		ai_chance = { base = 40 }
	}
}

# Dissolution: a restless member may leave the Union peacefully. root = member, scope:ck_eu_president = Union ruler
ck_situations.0110 = {
	type = character_event
	title = ck_situations.0110.t
	desc = ck_situations.0110.desc
	theme = diplomacy
	left_portrait = scope:ck_eu_president
	trigger = {
		liege ?= scope:ck_eu_president
		scope:ck_eu_president = { has_title = title:e_european_union }
	}
	option = {	# leave
		name = ck_situations.0110.a
		ck_eu_leave_union_effect = yes
		ck_fire_situation_catalyst_effect = { SITUATION = ck_european_question CATALYST = ck_catalyst_eu_member_left CHARACTER = root }
		ai_chance = {
			base = 40
			modifier = {
				add = 60
				joined_faction ?= {
					OR = {
						faction_is_type = independence_faction
						faction_is_type = liberty_faction
					}
				}
			}
			opinion_modifier = {
				opinion_target = scope:ck_eu_president
				multiplier = -1
			}
		}
	}
	option = {	# stay
		name = ck_situations.0110.b
		ai_chance = {
			base = 40
			opinion_modifier = { opinion_target = scope:ck_eu_president }
		}
	}
}
