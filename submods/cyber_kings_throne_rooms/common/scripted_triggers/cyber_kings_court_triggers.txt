# Cyber Kings court-scene triggers (gen_court_scene.py). Root = the court owner; by the ruler's culture,
# the same split as the clothing (cyber_kings_clothing_triggers.txt).

cyber_court_cywizard_trigger = {
	culture ?= {
		OR = {
			this = culture:archon
			this = culture:aeon
		}
	}
}

cyber_court_spacer_trigger = {
	culture ?= { has_cultural_pillar = heritage_ck_spacer }
}

cyber_court_earth_trigger = {
	culture ?= {
		NOR = {
			has_cultural_pillar = heritage_ck_spacer
			has_cultural_pillar = heritage_archaic
			has_cultural_pillar = heritage_proto
			has_cultural_pillar = heritage_cathrian
			has_cultural_pillar = heritage_denisovan
			has_cultural_pillar = heritage_ckw_selenic
			has_cultural_pillar = heritage_ckw_hermean
			has_cultural_pillar = heritage_ckw_utaridian
			has_cultural_pillar = heritage_ckw_stilbonian
			has_cultural_pillar = heritage_ckw_qairunan
			has_cultural_pillar = heritage_ckw_aphroditian
			has_cultural_pillar = heritage_ckw_hyndlan
			has_cultural_pillar = heritage_ckw_phoebean
			has_cultural_pillar = heritage_ckw_hestian
			has_cultural_pillar = heritage_ckw_ganisic
			has_cultural_pillar = heritage_ckw_venerian
			has_cultural_pillar = heritage_ckw_hesperian
			has_cultural_pillar = heritage_ckw_memnonian
			has_cultural_pillar = heritage_ckw_thaumasian
			has_cultural_pillar = heritage_ckw_alban
			has_cultural_pillar = heritage_ckw_isidian
			has_cultural_pillar = heritage_ckw_ahunan
			has_cultural_pillar = heritage_ckw_artemian
			has_cultural_pillar = heritage_ckw_asgardian
			has_cultural_pillar = heritage_ckw_valhallan
			has_cultural_pillar = heritage_ckw_tyrian
			has_cultural_pillar = heritage_ckw_agenorian
			has_cultural_pillar = heritage_ckw_galilean
			has_cultural_pillar = heritage_ckw_nergalian
			has_cultural_pillar = heritage_ckw_beletian
			has_cultural_pillar = heritage_ckw_kronian
			has_cultural_pillar = heritage_ckw_neptunian
			this = culture:lunar
			this = culture:mercurian
			this = culture:venusian
			this = culture:yonagunian
			this = culture:martian
		}
	}
}

cyber_court_any_trigger = {
	OR = {
		cyber_court_cywizard_trigger = yes
		cyber_court_spacer_trigger = yes
		cyber_court_earth_trigger = yes
	}
}
