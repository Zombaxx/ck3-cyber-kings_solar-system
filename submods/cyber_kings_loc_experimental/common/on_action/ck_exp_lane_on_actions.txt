# Cyber Kings experimental (2026-11-13): space-lane travel flavor events.
# Appends to the vanilla on_action without overriding it (method from vanilla common/on_action/_on_actions.info).
on_travel_plan_movement = {
	on_actions = { ck_lane_travel_on_action }
}

ck_lane_travel_on_action = {
	trigger = {
		is_ai = no # flavor only; saves AI performance
		highest_held_title_tier >= tier_county
		location ?= { geographical_region = ck_space_lanes }
		NOT = { has_character_flag = ck_lane_event_recent }
	}
	random_events = {
		chance_of_no_event = { value = 80 }
		100 = ck_lane.0001 # Aurora storm
		100 = ck_lane.0002 # Spacer Guild toll
		100 = ck_lane.0003 # Derelict in the lane
		100 = ck_lane.0004 # Looking into the Mare Incognitum
	}
}
