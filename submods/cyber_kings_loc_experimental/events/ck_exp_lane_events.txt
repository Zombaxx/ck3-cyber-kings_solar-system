# Cyber Kings experimental (2026-11-13): space-lane travel flavor events.
# Fired from ck_lane_travel_on_action (common/on_action/ck_exp_lane_on_actions.txt) while a player ruler travels a space lane.
namespace = ck_lane

ck_lane.0001 = {
	type = character_event
	title = ck_lane.0001.t
	desc = ck_lane.0001.desc
	theme = travel_danger
	override_background = { reference = fp1_ocean }
	left_portrait = {
		character = root
		animation = worry
	}
	cooldown = { years = 5 }
	trigger = {
		is_available_travelling_adult = yes
		is_playable_character = yes
		location ?= { geographical_region = ck_space_lanes }
	}
	immediate = {
		add_character_flag = { flag = ck_lane_event_recent years = 1 }
	}
	option = { # Wait for the storm to pass
		name = ck_lane.0001.a
		current_travel_plan ?= { delay_travel_plan = { days = 10 } }
		stress_impact = { impatient = minor_stress_impact_gain }
		ai_chance = { base = 50 }
	}
	option = { # Order the pilot through it
		name = ck_lane.0001.b
		random_list = {
			60 = {
				desc = ck_lane.0001.b.success
				add_prestige = minor_prestige_gain
			}
			40 = {
				desc = ck_lane.0001.b.failure
				add_stress = medium_stress_gain
				current_travel_plan ?= { delay_travel_plan = { days = 20 } }
			}
		}
		stress_impact = { craven = medium_stress_impact_gain }
		ai_chance = { base = 50 }
	}
}
