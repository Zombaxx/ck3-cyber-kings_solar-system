# Cyber Kings clothing triggers (gen_clothes_script.py). Used by portrait_modifiers/cyber_kings_clothes.txt.
# Character scope, scope:culture = the portrait culture (as vanilla portrait_shared_clothing_contents_trigger).

cyber_portrait_cywizard_clothing_trigger = {
	OR = {
		portrait_has_trait_trigger = { TRAIT = full_cybernetic_conversion }
		scope:culture ?= {
			OR = {
				this = culture:archon
				this = culture:aeon
			}
		}
	}
}

cyber_portrait_spacer_clothing_trigger = {
	scope:culture ?= { has_cultural_pillar = heritage_ck_spacer }
}
