# Cyber Kings: overrides vanilla. The paper map is a display aboard a Spacer Guild shuttle (they dock to the Aurora,
# the Guild's capital ship, between trips). Vanilla's medieval tables (ce1, ep3, tgp) are left out on purpose.
#
# A "Command Deck" (holo-table) = default: Spacers, Archons, Aeons, the old otherworlden peoples, and the
#   front end before a character is chosen. Key kept as vanilla's so any fallback to the western style lands here.
# B "Observation Wall" (wall screen + windows) = one style per home world, picked from the player's culture:
#   the shuttle is in orbit of that world. Culture -> world: the e46 world heritages (heritage_ckw_*) plus the
#   pre-split world cultures; every other heritage counts as Earth. Generated by _dev/scripts/map_table/gen_styles.py
map_table_style_western = {
	environment = "gfx/map/environment/environment_cyber_command_deck.txt"
	default = yes
}
