# Cyber Kings: overrides vanilla. The paper map is a display aboard a Spacer Guild shuttle (they dock to the Aurora,
# the Guild's capital ship, between trips). Vanilla's medieval tables (ce1, ep3, tgp) are left out on purpose.
#
# A "Command Deck" (holo-table) = default: Spacers, Archons, the old otherworlden peoples, and the front end before a
#   character is chosen. Key kept as vanilla's so any fallback to the western style lands here.
# C "Bridge Viewscreen" (hologram map over the flight console, panoramic window) = one style per home world, picked
#   from the player's culture; Earth has six faces (Europe/Africa default; the Americas; North America; South/East Asia;
#   Northeast Asia; Australia). Culture -> world:
#   the e46 world heritages (heritage_ckw_*) plus the pre-split world cultures; every other heritage counts as Earth.
#   Generated by _dev/scripts/map_table/gen_styles_c.py (style keys are still map_table_style_cyber_<key>).
map_table_style_western = {
	environment = "gfx/map/environment/environment_cyber_command_deck.txt"
	default = yes
}
