	type = language
	is_shown = {
		language_is_shown_trigger = {
			LANGUAGE = language_ck_inuit
		}
	}

		value = 10
		if = {
			limit = { has_cultural_pillar = language_ck_inuit }
			multiply = 10
		}
	}
