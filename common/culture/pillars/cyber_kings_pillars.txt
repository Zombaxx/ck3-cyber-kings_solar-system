language_archaic = {
	type = language
	is_shown = {
		language_is_shown_trigger = {
			LANGUAGE = language_archaic
		}
	}
	ai_will_do = {
		value = 10
		if = {
			limit = { has_cultural_pillar = language_archaic }
			multiply = 10
		}
	}
	color = { 1.0 1.0 0.5 }
}
