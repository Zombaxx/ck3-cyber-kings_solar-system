# Cyber Kings (2026-11-18): THE SOLAR TRADE ROUTE - booms and busts of trade along the space lanes and highways between

# (All Under Heaven) uses its own code window and map mode, which may need that DLC.

# Phases: Open Lanes (start) -> Trade Boom / Lane Piracy / Embargo -> Open Lanes. Wars between hub holders breed piracy,
# wars between great powers and Sovereign Ascendancy (Charter and Crown) breed embargoes, peace and Corporate Ascendancy
# breed booms.

	}

					takeover_points = 100

			}
			modifier_sets = {
				ck_st_lane_piracy_effects = {
					icon = "gfx/interface/icons/situations/situation_modifier_military.dds"
					ck_st_hub_holders = {
						character_modifier = { trade_capacity_mult = -0.15 }
					}
					ck_st_traders = {
						character_modifier = { trade_capacity_mult = -0.1 }

			illustration = "gfx/interface/illustrations/decisions/decision_legitimacy.dds"

			max_duration = { years = 20 }

			future_phases = {
				ck_st_open_lanes = {

						ck_catalyst_st_peace_year = 10
