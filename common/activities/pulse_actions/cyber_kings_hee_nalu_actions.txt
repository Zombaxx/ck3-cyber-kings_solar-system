apa_hee_nalu_sea_turtle = {
	icon = stress_loss

	is_valid = {
		is_current_phase_active = yes
		NOT = { has_variable = apa_hee_nalu_sea_turtle }
	}

	weight = {
		value = 10
	}

	effect = {
		random_attending_character = {
			save_scope_as = first
		}
		set_variable = apa_hee_nalu_sea_turtle
		add_activity_log_entry = {
			key = apa_hee_nalu_sea_turtle_key
			tags = { pulse_action }
			character = scope:first
			scope:first = {
				add_stress = minor_stress_loss
			}
		}
	}
}
