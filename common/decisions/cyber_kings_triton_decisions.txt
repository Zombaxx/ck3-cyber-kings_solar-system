# Cyber Kings (2026-11-17): Become Nomadic for the Tritonian Steppe (_dev/DLC_ADAPTATION.md section 20).
# Copy of vanilla 1.21 become_nomadic_decision (mpo_decisions.txt), which only lists vanilla steppe situations.
# Unlike vanilla, the AI may take it (rarely), so the nomad life can spread south on its own.
ck_triton_become_nomadic_decision = {
	picture = {
		reference = "gfx/interface/illustrations/event_story/mpo_steppe_region.dds"
	}
	desc = become_nomadic_decision_desc
	selection_tooltip = become_nomadic_decision_tooltip
	decision_group_type = realm
	sort_order = 40

	ai_check_interval = 120

	is_shown = {
		is_playable_character = yes
		NOR = {
			government_has_flag = government_is_steppe_admin
			government_has_flag = government_is_nomadic
		}
		capital_county ?= {
			any_county_situation = { this = situation:ck_triton_steppe }
		}
		has_mpo_dlc_trigger = yes
	}

	is_valid = {
		is_independent_ruler = yes
		is_landed = yes
		has_government = tribal_government
		realm_size < 10
	}

	is_valid_showing_failures_only = {
		is_at_war = no
		is_available = yes
	}

	cost = {
		prestige = {
			value = major_prestige_value
		}
	}

	effect = {
		every_sub_realm_county = {
			limit = {
				title_province = {
					has_holding_type = tribal_holding
				}
			}
			title_province = {
				set_holding_type = nomad_holding
			}
		}
		change_government = nomad_government
		every_vassal_or_below = {
			limit = {
				has_government = tribal_government
				highest_held_title_tier >= tier_county
			}
			change_government = nomad_government
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 5
		modifier = {	# neighbours already live as nomads
			add = 10
			any_neighboring_top_liege_realm_owner = { government_has_flag = government_is_nomadic }
		}
		modifier = { add = 5 has_trait = adventurer }
		modifier = { factor = 0 has_trait = content }
	}
}
