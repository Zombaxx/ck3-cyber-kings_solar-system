# Cyber Kings: tributaries and confederations set up at game start (2026-11-11).
# Appended to vanilla on_game_start (on_actions lists from several files are merged).
# Uses vanilla break_subject_contract_and_establish_tributary_effect (common/scripted_effects/00_tributary_setup_effects.txt).
on_game_start = {
	on_actions = { cyber_tributary_setup cyber_commonwealth_setup }
}

cyber_tributary_setup = {
	effect = {
		# Russia is a tributary of the Hegemony of Eurasia (Illuminious, the Cywizard)
		character:107175 ?= {
			save_scope_as = suzerain
			character:106215 ?= {
				if = {
					limit = { is_ruler = yes }
					break_subject_contract_and_establish_tributary_effect = { SUZERAIN = scope:suzerain TRIBUTARY = this }
				}
			}
		}
	}
}
