# Cyber Kings (2026-11-18): DLC fallbacks. The mod needs no DLC; content that uses a DLC mechanic switches on when the DLC is
# enabled and falls back to a base-game equivalent when it is not (_dev/DLC_ADAPTATION.md section 26).
# Vanilla does the same: history/titles/k_thessalonika.txt etc. turn administrative holders feudal when NOT roads_to_power, and
# common/on_action/game_start.txt turns nomads/herders tribal when has_mpo_dlc_trigger = no (that part already covers the mod's
# nomads, since the mod keeps vanilla's game_start.txt).
# Without the administrative-government feature the game refuses to let a player pick an administrative character
# (common/scripted_rules/00_rules.txt, GAME_OVER_CANNOT_PLAY_ADMIN), so the mod's administrative starts (Eurasia, the Galilean
# Moons, Ganymede, Kolob and their duchies/counties) become feudal for players without Roads to Power.
# on_game_start runs before the lobby, so the bookmark screen already shows them as feudal.
on_game_start = { on_actions = { ck_dlc_fallback_governments } }

ck_dlc_fallback_governments = {
	effect = {
		if = {
			limit = { NOT = { has_dlc_feature = admin_gov } }
			every_ruler = {
				limit = { government_has_flag = government_is_administrative }
				change_government = feudal_government
			}
		}
	}
}
