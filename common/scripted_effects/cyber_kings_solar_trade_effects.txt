# Cyber Kings (2026-11-18): effects for the Solar Trade Route (ck_solar_trade). _dev/DLC_ADAPTATION.md section 22.

# root = situation, yearly
ck_solar_trade_yearly_effect = {
	if = {
		limit = { situation_top_has_catalyst = ck_catalyst_st_passing_years }
		trigger_situation_catalyst = ck_catalyst_st_passing_years
	}
	if = {	# peace between the hubs
		limit = {
			situation_top_has_catalyst = ck_catalyst_st_peace_year
			NOT = {
				any_independent_ruler = {
					ck_st_hub_holder_trigger = yes
					any_primary_war_enemy = { ck_st_hub_holder_trigger = yes }
				}
			}
		}
		trigger_situation_catalyst = ck_catalyst_st_peace_year
	}
	if = {
		limit = {
			situation_top_has_catalyst = ck_catalyst_st_corporate_ascendancy
			situation:ck_corporate_wars ?= { situation_current_phase = ck_cc_corporate_ascendancy }
		}
		trigger_situation_catalyst = ck_catalyst_st_corporate_ascendancy
	}
	if = {
		limit = {
			situation_top_has_catalyst = ck_catalyst_st_sovereign_ascendancy
			situation:ck_corporate_wars ?= { situation_current_phase = ck_cc_sovereign_ascendancy }
		}
		trigger_situation_catalyst = ck_catalyst_st_sovereign_ascendancy
	}
}
