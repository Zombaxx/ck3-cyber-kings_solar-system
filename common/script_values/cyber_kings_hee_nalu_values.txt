# He'e Nalu (2026-11-15): surf skill as a number, added to the success weights in events/cyber_kings_hee_nalu_events.txt.
hee_nalu_surf_skill_value = {
	value = 0
	if = {
		limit = { has_trait = hee_nalu_kook }
		add = -15
	}
	else_if = {
		limit = { has_trait_xp = { trait = hee_nalu_surfer track = hee_nalu_surfer value >= 100 } }
		add = 30
	}
	else_if = {
		limit = { has_trait_xp = { trait = hee_nalu_surfer track = hee_nalu_surfer value >= 65 } }
		add = 20
	}
	else_if = {
		limit = { has_trait_xp = { trait = hee_nalu_surfer track = hee_nalu_surfer value >= 30 } }
		add = 10
	}
	else_if = {
		limit = { has_trait = hee_nalu_surfer }
		add = 5
	}
}
